1. Visión general del proyecto "RentAuto"

RentAuto es una alternativa didáctica al proyecto ATMC. El objetivo es construir un sistema de escritorio para Windows utilizando C# Windows Forms y una base de datos MySQL. Al igual que el cajero original, este proyecto mezcla dos dinámicas indispensables en la formación de un programador:

Capa con Base de Datos

Un Login seguro con roles y contraseña encriptada (BCrypt) y una ventana de administración (CRUD) para agregar, buscar, editar y borrar vehículos del inventario.

POO Pura (En Memoria)

Una simulación de renta de vehículos que utiliza herencia y polimorfismo para calcular tarifas según el tipo de auto (Compacto, Camioneta, Deportivo) sin persistencia física.

2. Base de Datos (MySQL)

Abre tu manejador de bases de datos preferido (MySQL Workbench, phpMyAdmin, DBeaver) y ejecuta el siguiente script. Creamos la base de datos rentauto_db, una tabla de roles, una tabla de usuarios que usará encriptación, y la tabla de vehículos para el CRUD.

Script_BaseDatos.sql MySQL 
-- 1. Crear base de datos
CREATE DATABASE rentauto_db;
USE rentauto_db;

-- 2. Crear tabla de roles
CREATE TABLE rol (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL
);

-- 3. Crear tabla de usuarios
CREATE TABLE usuario (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL, -- Almacenará el Hash de BCrypt
    rol_id INT NOT NULL,
    FOREIGN KEY (rol_id) REFERENCES rol(id)
);

-- 4. Crear tabla de vehículos (Para el CRUD)
CREATE TABLE vehiculo (
    id INT AUTO_INCREMENT PRIMARY KEY,
    marca VARCHAR(50) NOT NULL,
    modelo VARCHAR(50) NOT NULL,
    placa VARCHAR(15) NOT NULL UNIQUE,
    precio_dia DECIMAL(10,2) NOT NULL,
    disponible TINYINT(1) DEFAULT 1
);

-- 5. Insertar roles requeridos
INSERT INTO rol (nombre) VALUES ('Administrador'), ('Operador');

-- 6. Insertar usuario base (Contraseña en texto plano: "admin123")
-- El hash que ves abajo corresponde a la encriptación BCrypt de "admin123"

VALUES ('admin', '$2a$11$R9h/lS.U964l1y3Yf3sM3O8v0Gg0Yh9S76Q.6X8mC038a3q86Z1uO', 1);

Para hacer login por primera vez, utiliza el usuario admin y la contraseña admin123. La contraseña ya fue pre-encriptada con el algoritmo BCrypt para que encaje con la verificación que escribiremos en C#.

3. Conexión C# (Explicada a fondo)

Para interactuar con MySQL en C#, es indispensable instalar el paquete NuGet MySql.Data. A continuación, crearemos la clase clsConexion.cs que gestionará de forma segura el acceso a la base de datos.

Generador Dinámico de Cadena de Conexión (C#)

Cambia los parámetros para ver cómo se modifica la conexión en tiempo real.

Server=127.0.0.1; Database=rentauto_db; Uid=root; Pwd=; Port=3306;
Código de clsConexion.cs:
clsConexion.cs C# 
using System;
using MySql.Data.MySqlClient; // Requerido: Driver de conexión de MySQL

namespace RentAuto
{
    public class clsConexion
    {
        // Almacena la ruta de acceso a la Base de Datos
        private readonly string cadenaConexion;

        public clsConexion()
        {
            // Constructor: Inicializa la cadena de conexión
            // En un entorno real se recomienda cargarlo desde el archivo App.config
            cadenaConexion = "Server=127.0.0.1; Database=rentauto_db; Uid=root; Pwd=; Port=3306;";
        }

        // Método que intenta abrir la conexión y retornarla lista para usar
        public MySqlConnection ObtenerConexion()
        {
            try
            {
                MySqlConnection conexion = new MySqlConnection(cadenaConexion);
                conexion.Open(); // Intenta abrir la conexión física
                return conexion;
            }
            catch (Exception ex)
            {
                // Si falla (ejemplo: mysql apagado), retorna null para validar después
                Console.WriteLine("Error de conexión: " + ex.Message);
                return null;
            }
        }
    }
}

4. Módulo de Login Seguro (BCrypt)

Para el login, usaremos el paquete NuGet BCrypt.Net-Next. Esto evita que guardemos contraseñas visibles en la BD. En lugar de eso, comparamos el "Hash" de seguridad con la contraseña en texto plano introducida por el usuario usando BCrypt.Verify.

frmLogin.cs (Lógica del Botón Iniciar Sesión) C# 
using System;
using System.Windows.Forms;
using MySql.Data.MySqlClient;
using BCrypt.Net; // Requerido: Librería para hashing de seguridad de contraseñas

namespace RentAuto
{
    public partial class frmLogin : Form
    {
        public frmLogin()
        {
            InitializeComponent();
        }

        private void btnIniciar_Click(object sender, EventArgs e)
        {
            string usuario = txtUser.Text.Trim();
            string contrasena = txtPassword.Text.Trim();

            if (string.IsNullOrEmpty(usuario) || string.IsNullOrEmpty(contrasena))
            {
                MessageBox.Show("Por favor, llena todos los campos.", "Campos vacíos", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            clsConexion conClase = new clsConexion();
            MySqlConnection dbConn = conClase.ObtenerConexion();

            if (dbConn == null)
            {
                MessageBox.Show("No se pudo conectar a la Base de Datos. Verifica que MySQL esté activo.", "Error de Red", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return;
            }

            // Usamos consulta parametrizada para prevenir Inyección SQL
            string query = "SELECT password, rol_id FROM usuario WHERE nombre = @nombre";
            MySqlCommand cmd = new MySqlCommand(query, dbConn);
            cmd.Parameters.AddWithValue("@nombre", usuario);

            MySqlDataReader reader = cmd.ExecuteReader();

            if (reader.Read())
            {
                string hashAlmacenado = reader["password"].ToString();
                int rolId = Convert.ToInt32(reader["rol_id"]);

                // Comparamos de forma segura la contraseña escrita con el Hash de la BD
                bool credencialValida = BCrypt.Net.BCrypt.Verify(contrasena, hashAlmacenado);

                if (credencialValida)
                {
                    MessageBox.Show("¡Bienvenido al sistema!", "Acceso Autorizado", MessageBoxButtons.OK, MessageBoxIcon.Information);
                    dbConn.Close();

                    this.Hide(); // Ocultar el login actual
                    frmPrincipal pantallaMdi = new frmPrincipal(usuario, rolId);
                    pantallaMdi.Show(); // Abrir el panel principal
                }
                else
                {
                    MessageBox.Show("Contraseña incorrecta.", "Error de autenticación", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
            else
            {
                MessageBox.Show("El usuario no existe.", "Error de autenticación", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }

            dbConn.Close(); // Siempre cerrar las conexiones abiertas
        }
    }
}

5. Módulo CRUD de Vehículos (frmVehiculos.cs)

Este es el núcleo de tu interacción con los datos. Contiene los métodos para Leer (Read) usando un adaptador para llenar una tabla visual (DataGridView), y Crear, Actualizar y Eliminar con consultas directas parametrizadas.

frmVehiculos.cs C# 
using System;
using System.Data;
using System.Windows.Forms;
using MySql.Data.MySqlClient;

namespace RentAuto
{
    public partial class frmVehiculos : Form
    {
        private clsConexion dbHelper = new clsConexion();

        public frmVehiculos()
        {
            InitializeComponent();
            CargarTabla(); // Cargar datos al abrir la ventana
        }

        // READ: Llena el DataGridView con los vehículos de la base de datos
        private void CargarTabla()
        {
            MySqlConnection conn = dbHelper.ObtenerConexion();
            if (conn != null)
            {
                string sql = "SELECT id, marca, modelo, placa, precio_dia, disponible FROM vehiculo";
                MySqlDataAdapter adapter = new MySqlDataAdapter(sql, conn);
                DataTable dataTable = new DataTable();
                adapter.Fill(dataTable); // Llena el contenedor en memoria
                dgvVehiculos.DataSource = dataTable; // Vincula los datos a la tabla visual
                conn.Close();
            }
        }

        // CREATE: Registra un nuevo vehículo
        private void btnGuardar_Click(object sender, EventArgs e)
        {
            MySqlConnection conn = dbHelper.ObtenerConexion();
            if (conn == null) return;

            string sql = "INSERT INTO vehiculo (marca, modelo, placa, precio_dia) VALUES (@marca, @modelo, @placa, @precio)";
            MySqlCommand cmd = new MySqlCommand(sql, conn);
            cmd.Parameters.AddWithValue("@marca", txtMarca.Text);
            cmd.Parameters.AddWithValue("@modelo", txtModelo.Text);
            cmd.Parameters.AddWithValue("@placa", txtPlaca.Text);
            cmd.Parameters.AddWithValue("@precio", Convert.ToDecimal(txtPrecio.Text));

            int filas = cmd.ExecuteNonQuery(); // Devuelve las filas afectadas
            conn.Close();

            if (filas > 0)
            {
                MessageBox.Show("Vehículo registrado con éxito.", "Guardar", MessageBoxButtons.OK, MessageBoxIcon.Information);
                CargarTabla(); // Actualizar la tabla visual
            }
        }

        // DELETE: Elimina un vehículo por su ID
        private void btnEliminar_Click(object sender, EventArgs e)
        {
            if (dgvVehiculos.SelectedRows.Count == 0) return;

            int idSeleccionado = Convert.ToInt32(dgvVehiculos.SelectedRows[0].Cells["id"].Value);

            DialogResult confirm = MessageBox.Show("¿Seguro que deseas eliminar este vehículo?", "Confirmación", MessageBoxButtons.YesNo, MessageBoxIcon.Question);
            if (confirm == DialogResult.No) return;

            MySqlConnection conn = dbHelper.ObtenerConexion();
            if (conn == null) return;

            string sql = "DELETE FROM vehiculo WHERE id = @id";
            MySqlCommand cmd = new MySqlCommand(sql, conn);
            cmd.Parameters.AddWithValue("@id", idSeleccionado);

            cmd.ExecuteNonQuery();
            conn.Close();
            CargarTabla();
        }
    }
}

6. Clases POO de Renta (Polimorfismo en Memoria)

A diferencia del CRUD de vehículos, para la parte práctica de Programación Orientada a Objetos (POO), simularemos el cálculo de costos de renta en memoria con un diseño robusto utilizando una Clase Abstracta y clases derivadas (herencia) que implementan reglas de cálculo dinámicas (polimorfismo).

VehiculoRenta.cs (Estructura POO de clases) C# 
using System;

namespace RentAuto
{
    // Clase Abstracta Base (No se puede instanciar directamente, funciona como plantilla)
    public abstract class VehiculoRenta
    {
        protected string placa;
        protected float costoBaseDia;

        public string Placa { get { return placa; } set { placa = value; } }
        public float CostoBaseDia { get { return costoBaseDia; } }

        public VehiculoRenta(string placa, float costoBaseDia)
        {
            this.placa = placa;
            this.costoBaseDia = costoBaseDia;
        }

        // Método abstracto: Las clases hijas DEBEN implementar su propio cálculo de costos
        public abstract float CalcularCostoRenta(int dias);
    }

    // Clase Hija 1: Auto Compacto (Mantiene costo base)
    public class AutoCompacto : VehiculoRenta
    {
        public AutoCompacto(string placa, float costoBase) : base(placa, costoBase) { }

        public override float CalcularCostoRenta(int dias)
        {
            return costoBaseDia * dias; // Costo directo sin cargos adicionales
        }
    }

    // Clase Hija 2: Camioneta SUV (Tiene un cargo adicional fijo por seguro contra todo terreno)
    public class CamionetaSUV : VehiculoRenta
    {
        private float cargoSeguroFijo = 150.00f; // Atributo exclusivo de la camioneta

        public CamionetaSUV(string placa, float costoBase, float seguro) : base(placa, costoBase)
        {
            this.cargoSeguroFijo = seguro;
        }

        // Sobrecarga / Polimorfismo: Calcula el costo sumando el seguro obligatorio
        public override float CalcularCostoRenta(int dias)
        {
            return (costoBaseDia * dias) + cargoSeguroFijo;
        }
    }
}

7. Configuración del Formulario Principal MDI

Para darle un aspecto profesional a la aplicación, configuramos el formulario principal (frmPrincipal.cs) como un contenedor MDI (Multiple Document Interface). Esto permite incrustar los subformularios dentro de su espacio de trabajo utilizando la propiedad MdiParent, evitando ventanas flotantes caóticas.

Pasos previos en el Diseñador de Visual Studio:

  1. Crea un nuevo formulario y llámalo frmPrincipal.
  2. En su panel de propiedades, busca IsMdiContainer y configúralo en True.
  3. Agrega un control MenuStrip en la parte superior para organizar los botones de acceso ("Gestión de autos", "Simulador Renta").
frmPrincipal.cs (Control de Navegación del Sistema) C# 
using System;
using System.Windows.Forms;

namespace RentAuto
{
    public partial class frmPrincipal : Form
    {
        // Instancias para evitar que se abran copias duplicadas de la misma ventana
        private frmVehiculos ventanaCrud;

        private string usuarioActivo;
        private int rolId;

        public frmPrincipal(string usuario, int rol)
        {
            InitializeComponent();
            usuarioActivo = usuario;
            this.rolId = rol;

            // Personalizar el título de la ventana con el usuario que inició sesión
            this.Text = "Panel de Control RentAuto - Usuario: " + usuarioActivo;
        }

        // Evento que se dispara al pulsar la opción de "Gestión de autos" en el menú superior
        private void tsmVehiculos_Click(object sender, EventArgs e)
        {
            // Patrón para verificar si el formulario ya está activo o fue cerrado (Disposed)
            if (ventanaCrud == null || ventanaCrud.IsDisposed)
            {
                ventanaCrud = new frmVehiculos();
                ventanaCrud.MdiParent = this; // Indicamos que debe vivir DENTRO de frmPrincipal
                ventanaCrud.Show();
            }
            else
            {
                ventanaCrud.BringToFront(); // Si ya existe, lo trae al frente en vez de duplicarlo
            }
        }
    }
}
